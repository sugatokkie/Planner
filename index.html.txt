    <script>
        const todoForm = document.getElementById('todo-form');
        const todoInput = document.getElementById('todo-input');
        const dateInput = document.getElementById('date-input');
        const imageInput = document.getElementById('image-input');
        const todoList = document.getElementById('todo-list');

        // 페이지 로드 시 기존 저장된 데이터 불러오기
        document.addEventListener('DOMContentLoaded', loadTodos);

        todoForm.addEventListener('submit', function(e) {
            e.preventDefault();

            const text = todoInput.value;
            const date = dateInput.value;
            const file = imageInput.files[0];

            if (file) {
                const reader = new FileReader();
                reader.onload = function(e) {
                    const imageData = e.target.result;
                    addTodoItem(text, date, imageData, false);
                    saveTodo(text, date, imageData, false);
                };
                reader.readAsDataURL(file);
            } else {
                addTodoItem(text, date, null, false);
                saveTodo(text, date, null, false);
            }

            // 입력창 초기화
            todoInput.value = '';
            dateInput.value = '';
            imageInput.value = '';
        });

        function addTodoItem(text, date, imageData, isCompleted) {
            const li = document.createElement('li');

            const itemLeft = document.createElement('div');
            itemLeft.className = 'item-left';

            const checkbox = document.createElement('input');
            checkbox.type = 'checkbox';
            checkbox.checked = isCompleted;

            const textSpan = document.createElement('span');
            textSpan.textContent = text;
            if (isCompleted) textSpan.classList.add('completed');

            checkbox.addEventListener('change', function() {
                textSpan.classList.toggle('completed');
                updateLocalStorage();
            });

            itemLeft.appendChild(checkbox);

            if (imageData) {
                const img = document.createElement('img');
                img.src = imageData;
                img.className = 'img-preview';
                itemLeft.appendChild(img);
            }

            itemLeft.appendChild(textSpan);

            if (date) {
                const dateBadge = document.createElement('span');
                dateBadge.className = 'date-badge';
                dateBadge.textContent = date;
                itemLeft.appendChild(dateBadge);
            }

            const deleteBtn = document.createElement('button');
            deleteBtn.textContent = '삭제';
            deleteBtn.className = 'delete-btn';
            deleteBtn.addEventListener('click', function() {
                li.remove();
                updateLocalStorage();
            });

            li.appendChild(itemLeft);
            li.appendChild(deleteBtn);
            todoList.appendChild(li);
        }

        // 브라우저 LocalStorage 저장
        function saveTodo(text, date, imageData, isCompleted) {
            const todos = JSON.parse(localStorage.getItem('myTodos') || '[]');
            todos.push({ text, date, imageData, isCompleted });
            localStorage.setItem('myTodos', JSON.stringify(todos));
        }

        function loadTodos() {
            const todos = JSON.parse(localStorage.getItem('myTodos') || '[]');
            todos.forEach(todo => addTodoItem(todo.text, todo.date, todo.imageData, todo.isCompleted));
        }

        function updateLocalStorage() {
            const todos = [];
            const items = todoList.querySelectorAll('li');
            items.forEach(li => {
                const text = li.querySelector('.item-left span:not(.date-badge)').textContent;
                const dateBadge = li.querySelector('.date-badge');
                const date = dateBadge ? dateBadge.textContent : '';
                const img = li.querySelector('.img-preview');
                const imageData = img ? img.src : null;
                const isCompleted = li.querySelector('input[type="checkbox"]').checked;

                todos.push({ text, date, imageData, isCompleted });
            });
            localStorage.setItem('myTodos', JSON.stringify(todos));
        }
    </script>
